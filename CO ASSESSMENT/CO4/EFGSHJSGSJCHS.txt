#include <stdio.h>

#define MAX 1000

int main()
{
    int disk[MAX] = {0};
    int n, fileSize, method;
    int i, j, start, count, indexBlock;

    printf("Enter total disk size (number of blocks): ");
    scanf("%d", &n);

    printf("Enter file size (number of blocks): ");
    scanf("%d", &fileSize);

    printf("\nFile Allocation Methods\n");
    printf("1. Contiguous Allocation\n");
    printf("2. Linked Allocation\n");
    printf("3. Indexed Allocation\n");
    printf("Enter allocation method: ");
    scanf("%d", &method);

    if (fileSize > n || fileSize <= 0)
    {
        printf("Error: Invalid file size.\n");
        return 0;
    }

    /* Contiguous Allocation */
    if (method == 1)
    {
        int found = 0;

        for (i = 0; i <= n - fileSize; i++)
        {
            count = 0;

            for (j = i; j < i + fileSize; j++)
            {
                if (disk[j] == 0)
                    count++;
                else
                    break;
            }

            if (count == fileSize)
            {
                start = i;
                found = 1;
                break;
            }
        }

        if (found)
        {
            printf("\nContiguous Allocation successful.\n");
            printf("Allocated blocks: ");

            for (i = start; i < start + fileSize; i++)
            {
                disk[i] = 1;
                printf("%d ", i);
            }
            printf("\n");
        }
        else
        {
            printf("\nError: Contiguous allocation failed.\n");
            printf("Sufficient continuous free space is not available.\n");
        }
    }

    /* Linked Allocation */
    else if (method == 2)
    {
        count = 0;

        printf("\nLinked Allocation successful.\n");
        printf("Allocated blocks: ");

        for (i = 0; i < n && count < fileSize; i++)
        {
            if (disk[i] == 0)
            {
                disk[i] = 1;
                printf("%d ", i);
                count++;
            }
        }

        if (count < fileSize)
            printf("\nError: Not enough free blocks.\n");
        else
            printf("\n");
    }

    /* Indexed Allocation */
    else if (method == 3)
    {
        if (fileSize + 1 > n)
        {
            printf("\nError: Not enough blocks for indexed allocation.\n");
            return 0;
        }

        /* Find index block */
        indexBlock = -1;

        for (i = 0; i < n; i++)
        {
            if (disk[i] == 0)
            {
                indexBlock = i;
                disk[i] = 1;
                break;
            }
        }

        if (indexBlock == -1)
        {
            printf("\nError: No free index block available.\n");
            return 0;
        }

        count = 0;

        printf("\nIndexed Allocation successful.\n");
        printf("Index block: %d\n", indexBlock);
        printf("Allocated blocks: ");

        for (i = 0; i < n && count < fileSize; i++)
        {
            if (disk[i] == 0)
            {
                disk[i] = 1;
                printf("%d ", i);
                count++;
            }
        }

        if (count < fileSize)
        {
            printf("\nError: Not enough free blocks.\n");
        }
        else
        {
            printf("\n");
        }
    }

    else
    {
        printf("\nError: Invalid allocation method.\n");
    }

    return 0;
}
